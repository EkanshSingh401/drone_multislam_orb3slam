cmake_minimum_required(VERSION 3.16)
project(active_slam_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(active_slam_msgs REQUIRED)
find_package(active_slam_information REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(octomap REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time)

add_executable(octomap_mapper src/octomap_mapper.cpp)
target_include_directories(octomap_mapper PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(octomap_mapper Eigen3::Eigen)
target_link_libraries(octomap_mapper ${OCTOMAP_LIBRARIES})
ament_target_dependencies(octomap_mapper rclcpp sensor_msgs nav_msgs std_msgs)

add_executable(exploration_planner src/exploration_planner.cpp)
target_include_directories(exploration_planner PRIVATE include)
target_link_libraries(exploration_planner active_slam_information::active_slam_information Eigen3::Eigen ${OpenCV_LIBRARIES} ${Boost_LIBRARIES})
ament_target_dependencies(exploration_planner rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs active_slam_msgs ov_core ov_init ov_msckf)

add_executable(test_coverage_gain test/test_coverage_gain.cpp)
target_include_directories(test_coverage_gain PRIVATE include)
target_link_libraries(test_coverage_gain Eigen3::Eigen)
install(TARGETS octomap_mapper exploration_planner test_coverage_gain DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/offboard_executor.py scripts/vio_fault_injector.py scripts/vio_health_gate.py DESTINATION lib/${PROJECT_NAME})
ament_package()
