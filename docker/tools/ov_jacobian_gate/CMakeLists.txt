cmake_minimum_required(VERSION 3.16)
project(ov_jacobian_gate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(active_slam_msgs REQUIRED)
find_package(ov_core REQUIRED)
find_package(ov_init REQUIRED)
find_package(ov_msckf REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS system filesystem thread date_time)
# active_slam_information is compiled in directly (plain-CMake library, Eigen only)
set(ASI_DIR "" CACHE PATH "path to the active_slam_information source tree")
add_executable(ov_jacobian_gate src/ov_jacobian_gate.cpp
  ${ASI_DIR}/src/information_gain.cpp ${ASI_DIR}/src/view_geometry.cpp)
target_include_directories(ov_jacobian_gate PRIVATE ${ASI_DIR}/include ${EIGEN3_INCLUDE_DIR})
ament_target_dependencies(ov_jacobian_gate rclcpp rosbag2_cpp active_slam_msgs ov_core ov_init ov_msckf)
target_link_libraries(ov_jacobian_gate ${OpenCV_LIBRARIES} ${Boost_LIBRARIES})
install(TARGETS ov_jacobian_gate DESTINATION lib/${PROJECT_NAME})
ament_package()
